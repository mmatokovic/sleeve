{{- $path := .path -}} {{- $page := .page -}} {{- $isLocal := not (urls.Parse $path).Scheme -}} {{- $isPage := and (eq $page.Kind "page") (not $page.BundleType) -}} {{- $startsWithSlash := hasPrefix $path "/" -}} {{- $startsWithRelative := hasPrefix $path "../" -}} {{- if and $path $isLocal -}} {{- if $startsWithSlash -}} {{- $path = (relURL (strings.TrimPrefix "/" $path)) -}} {{- else if and $isPage (not $startsWithRelative) -}} {{ $path = (printf "../%s" $path) }} {{- end -}} {{- end -}} {{- return $path -}}
